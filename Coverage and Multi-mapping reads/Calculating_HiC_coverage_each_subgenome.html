Calculate coverge estimate for genome and subgenomes using valid read pairs

To calculate the coverage for the whole genome, we used the following formula:

Total coverage: \(Coverage_{total} = \frac{\text{Total number of valid pairs} \times 2 \times \text{Read length}}{\text{Genome size}}\).

To obtain a rough estimate of coverage for each subgenome, we used the following formula:

Coverage per subgenome: \(Coverage_{subgenome} = \frac{\text{# valid read pairs} \times 2 \times \text{Read length (bp)}}{\text{Subgenome size (bp)}}\)

# Read length for all species
read_length <- 146

Filtering

Filter out scaffolds from A.rouxii

# Filter out all named scaffold, only keeping the ones named SUPER
coverage_AmyRoux <- coverage_AmyRoux %>%
  filter(
    !grepl("scaffold", contig, ignore.case = TRUE) & 
    !grepl("MT", contig, ignore.case = TRUE) # Filter out MT
  )

Filter outsmall scaffolds in M.flavus

# Start by merging the windowed coverage file with the chromosome sizes file, using contig as "merging factor". 
merged_MucFlav <- left_join(coverage_MucFlav, MucFlav_sizes, by = "contig")
# Using the sizes column we filter out all contigs smaller than 100 000, removing scaffolds. 
filtered_MucFlav <- merged_MucFlav %>% filter(size >= 100000)

Filter out small scaffolds in M.racemosus

# Start by merging the windowed coverage file with the chromosome sizes file, using contig as "merging factor". 
merged_MucRace <- left_join(coverage_MucRace, MucRace_sizes, by = "contig")
# Filter out scaffolds smaller than 100 000.
filtered_MucRace1 <- merged_MucRace %>% filter(size >= 100000)

Filter out small scaffolds in M.racemosus

We do the same for M. racemosus as we did for M.flavus.

merged_MucRace <- left_join(coverage_MucRace, MucRace_sizes, by = "contig")
# Filter out scaffolds smaller than 100 000.
filtered_MucRace1 <- merged_MucRace %>% filter(size >= 100000)

Mean coverage per chromosome

# Find mean coverage per chromosome by grouping by contig and then use summarise to find the mean coverage for that chromosome 
coverage_summary_AmyRoux <- coverage_AmyRoux  %>%
  group_by(contig) %>%
  summarise(mean_coverage = mean(coverage))
coverage_summary_MucFlav <- filtered_MucFlav %>%
  group_by(contig) %>%
  summarise(mean_coverage = mean(coverage))
coverage_summary_MucRace <- filtered_MucRace1 %>%
  group_by(contig) %>%
  summarise(mean_coverage = mean(coverage))

Shorten chromosome names and add subgenome coloumn

# Shorten the chromosome name so it only includes chromosome number
# Add column indicating subgenome
coverage_AmyRoux_clean <- coverage_summary_AmyRoux %>% 
  # Create subgenome column. Classify based on the part before the first "_". 
  mutate(subgenome = str_extract(contig, "^[^_]+")) %>%
  # Remove everything before the last "_", only keeping the chromosome number
  mutate(contig_short = str_extract(contig, "[^_]+$")) %>% 
  mutate(subgenome = str_replace(subgenome, "subgenom(\\d+)", "subgenome\\1"))
# Shorten the chromosome name so it only includes chromosome number
# Add column indicating subgenome
coverage_MucFlav_clean <- coverage_summary_MucFlav %>% 
  # Create subgenome column. Classify based on the part before the first "_". 
  mutate(subgenome = str_extract(contig, "^[^_]+")) %>%
  # Remove everything before the last "_", only keeping the chromosome number
  mutate(contig_short = str_extract(contig, "[^_]+$")) 
# Shorten the chromosome name so it only includes chromosome number
# Add column indicating subgenome
coverage_MucRace_clean <- coverage_summary_MucRace %>% 
    # Create subgenome column based on first part before first "_"
  mutate(subgenome = str_extract(contig, "^[^_]+")) %>%
  # Remove everything before the last "_"
  mutate(contig_short = str_extract(contig, "[^_]+$"))

Arrange chromosomes and subgenomes before plotting

# Arrange so the numbers are shown by increasing number in all species
coverage_AmyRoux_clean <- coverage_AmyRoux_clean %>%
  # Sort by subgenome then chromosome
  arrange(subgenome, as.numeric(contig_short)) %>%   
  # unique for factor
  mutate(
    contig_unique = paste(subgenome, contig_short, sep = "_"),  
    contig_unique = factor(contig_unique, levels = contig_unique)
  )

coverage_MucFlav_clean <- coverage_MucFlav_clean %>%
  arrange(subgenome, as.numeric(contig_short)) %>%
  mutate(contig_short = factor(contig_short, levels = unique(contig_short)))

coverage_MucRace_clean <- coverage_MucRace_clean %>%
  arrange(subgenome, as.numeric(contig_short)) %>%
  mutate(contig_short = factor(contig_short, levels = unique(contig_short)))
